Respaldo & Configuración de Almacenamiento

La plataforma almacena todos tus bots, prospectos del CRM, catálogo de productos y registros de forma redundante y protegida en la nube.

Estructura de tablas y base de datos relacional para réplicas en la nube: 
-- TABLA DE BOTS CON FACETAS
create table if not exists bots (
  id text primary key,
  name text not null,
  phone text,
  prefix text default '#',
  status text default 'offline',
  qr_code text,
  pairing_code text,
  is_active boolean default true,
  personality text default 'recreational',
  system_prompt text,
  ai_auto_reply boolean default false,
  settings jsonb default '{}'::jsonb,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- TABLA DE LOGS DE ACTIVIDAD
create table if not exists bot_logs (
  id bigserial primary key,
  bot_id text references bots(id) on delete cascade,
  type text not null,
  chat_jid text,
  sender_jid text,
  sender_name text,
  content text,
  details jsonb default '{}'::jsonb,
  created_at timestamptz default now()
);

-- TABLA DE FUNCIONES Y REGLAS
create table if not exists bot_custom_functions (
  id text primary key,
  bot_id text default 'all',
  name text not null,
  description text,
  trigger_type text default 'command',
  trigger_value text not null,
  response_text text not null,
  response_type text default 'text',
  is_enabled boolean default true,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- TABLA DE PROSPECTOS CRM (LEADS)
create table if not exists bot_leads (
  id text primary key,
  bot_id text references bots(id) on delete cascade,
  name text,
  phone text,
  email text,
  interest text,
  notes text,
  status text default 'nuevo',
  created_at timestamptz default now()
);

-- TABLA DE CATÁLOGO DE PRODUCTOS
create table if not exists bot_products (
  id text primary key,
  bot_id text references bots(id) on delete cascade,
  name text not null,
  description text,
  price numeric default 0,
  currency text default 'USD',
  image_url text,
  category text default 'General',
  is_active boolean default true,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

alter table bots enable row level security;

alter table bot_products enable row level security;
